232 void materializeRoute(std::vector<uint16_t> &route) const {
233 if (parent == nullptr) {
234 const auto &stored_route = getRoute();
235 route.assign(stored_route.begin(), stored_route.end());
236 return;
237 }
238
239 const Label *root = this;
240 std::size_t depth = 0;
241 while (root->parent != nullptr) {
242 ++depth;
243 root = root->parent;
244 }
245
246 const auto &root_route = root->getRoute();
247 const std::size_t prefix_size = root_route.size();
248 route.resize(prefix_size + depth);
249 std::copy(root_route.begin(), root_route.end(), route.begin());
250
251 const Label *current = this;
252 std::size_t write = route.size();
253 while (current->parent != nullptr) {
254 route[--write] = static_cast<uint16_t>(current->node_id);
255 current = current->parent;
256 }
257 }
