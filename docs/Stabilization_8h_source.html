111 void project_onto_dual_domain(DualSolution &duals) const {
112 const std::size_t count = std::min(duals.size(), constraint_senses.size());
113 for (std::size_t i = 0; i < count; ++i) {
114 // BALDES solves a minimization master. Its '<' row duals are
115 // non-positive, '>' row duals are non-negative, and equality-row
116 // duals are unrestricted.
117 if (constraint_senses[i] == '<') {
118 duals[i] = std::min(0.0, duals[i]);
119 } else if (constraint_senses[i] == '>') {
120 duals[i] = std::max(0.0, duals[i]);

165 double _misprice_schedule(int nb_misprices, double base_alpha) {
166 // Table 1 (left): alpha_tilde = [1 - k * (1 - alpha)]_+
167 const int k = std::max(1, nb_misprices);
168 double alpha = std::max(0.0, 1.0 - k * (1.0 - base_alpha));
169 if (nb_misprices > 20 || alpha <= kAlphaEps) {
170 alpha = 0.0; // Deactivate stabilization
171 }
172 duals_in = duals_sep;
173 return alpha;

187 DualSolution getStabDualSol(const DualSolution &input_duals) {
188 if (input_duals.empty()) { return input_duals; }
189 align_with_master_duals(input_duals);
190 DualSolution pi_out = input_duals;
191 if (cur_stab_center.empty()) { return pi_out; }
192 if (cur_alpha <= 0.0) {
193 smooth_dual_sol = pi_out;
194 duals_sep = pi_out;
195 return pi_out;
196 }
197
198 // Non-directional smoothing:
199 // pi_tilde = alpha * pi_in + (1-alpha) * pi_out
200 const size_t n = pi_out.size();
201 DualSolution pi_tilde(n);
202 for (size_t i = 0; i < n; ++i) { pi_tilde[i] = cur_alpha * cur_stab_center[i] + (1.0 - cur_alpha) * pi_out[i]; }
203
204 // If directional components are unavailable, return the convex
205 // combination. Since both endpoints are dual-feasible, the combination
206 // respects each row's dual domain.
207 if (subgradient.empty() || subgradient_norm <= EPSILON || beta <= 0.0) {
208 smooth_dual_sol = pi_tilde;
209 duals_sep = pi_tilde;
210 return pi_tilde;
211 }
212
213 // Directional smoothing (Table 2):
214 // pi_g = pi_in + (g_in / ||g_in||) * ||pi_out - pi_in||
215 // rho = beta*pi_g + (1-beta)*pi_out
216 // pi_sep = (pi_in + ||pi_tilde-pi_in||/||rho-pi_in|| * (rho-pi_in))_+
217 const double norm_in_out = std::sqrt(std::inner_product(pi_out.begin(), pi_out.end(), cur_stab_center.begin(),
218 0.0, std::plus<double>(),
219 [](double a, double b) {
220 const double d = a - b;
221 return d * d;
222 }) +
223 EPSILON);
224
225 DualSolution pi_g(n);
226 for (size_t i = 0; i < n; ++i) {
227 pi_g[i] = cur_stab_center[i] + (subgradient[i] / subgradient_norm) * norm_in_out;
228 }
229
230 DualSolution rho(n);
231 for (size_t i = 0; i < n; ++i) { rho[i] = beta * pi_g[i] + (1.0 - beta) * pi_out[i]; }
232
233 const double norm_tilde_in = std::sqrt(std::inner_product(pi_tilde.begin(), pi_tilde.end(),
234 cur_stab_center.begin(), 0.0, std::plus<double>(),
235 [](double a, double b) {
236 const double d = a - b;
237 return d * d;
238 }) +
239 EPSILON);
240 const double norm_rho_in =
241 std::sqrt(std::inner_product(rho.begin(), rho.end(), cur_stab_center.begin(), 0.0, std::plus<double>(),
242 [](double a, double b) {
243 const double d = a - b;
244 return d * d;
245 }) +
246 EPSILON);
247
248 DualSolution pi_sep(n);
249 for (size_t i = 0; i < n; ++i) {
250 pi_sep[i] = cur_stab_center[i] + (norm_tilde_in / norm_rho_in) * (rho[i] - cur_stab_center[i]);
251 }
252 project_onto_dual_domain(pi_sep);
253 smooth_dual_sol = pi_sep;
254 duals_sep = pi_sep;
255 return pi_sep;
256 }

277 bool dynamic_alpha_schedule(const ModelData &dados) {
278 constexpr double DOT_TOLERANCE = 1e-3;
279 const size_t n = cur_stab_center.size();
280
281 // Compute relative distance: ||smooth_dual_sol - cur_stab_center|| /
282 // |lp_obj|
283 double rel_distance = norm(smooth_dual_sol, cur_stab_center) / (std::abs(lp_obj) + EPSILON);
284 if (rel_distance < kNormTolerance) {
285 alpha = cur_alpha = 0.0;
286 return false;
287 }
288
289 // Compute the difference vector (direction = smooth_dual_sol -
290 // cur_stab_center)
291 std::vector<double> direction(n);
292 for (size_t i = 0; i < n; ++i) { direction[i] = smooth_dual_sol[i] - cur_stab_center[i]; }
293
294 // Compute norm of direction vector using inner_product
295 double dir_norm =
296 std::sqrt(std::inner_product(direction.begin(), direction.end(), direction.begin(), 0.0) + EPSILON);
297 if (dir_norm < EPSILON || subgradient_norm < EPSILON) { return false; }
298
299 // Normalize the direction and the subgradient vectors.
300 std::vector<double> normalized_direction(n);
301 std::vector<double> normalized_subgradient(n);
302 for (size_t i = 0; i < n; ++i) {
303 normalized_direction[i] = direction[i] / dir_norm;
304 normalized_subgradient[i] = subgradient[i] / subgradient_norm;
305 }
306
307 // Compute the cosine of the angle between normalized_direction and
308 // normalized_subgradient.
309 double cos_angle = std::inner_product(normalized_direction.begin(), normalized_direction.end(),
310 normalized_subgradient.begin(), 0.0);
311
312 // If the cosine of the angle is very close to zero, then the vectors
313 // are nearly orthogonal.
314 return cos_angle < DOT_TOLERANCE;
315 }

317 void update_subgradient(const ModelData &model, const DualSolution &node_duals,
318 const std::vector<Label *> &pricing_columns) {
319 const std::size_t number_of_rows = node_duals.size();
320 if (model.b.size() != number_of_rows || model.sense.size() != number_of_rows) {
321 throw std::invalid_argument("Master RHS (" + std::to_string(model.b.size()) + "), senses (" +
322 std::to_string(model.sense.size()) + "), and dual solution (" +
323 std::to_string(number_of_rows) + ") have inconsistent row counts");
324 }
325
326 // Without pricing columns the Lagrangian solution is unknown, so
327 // there is no direction to adapt alpha or beta with.
328 if (pricing_columns.empty()) {
329 subgradient.clear();
330 subgradient_norm = 0.0;
331 return;
332 }
333
334 new_rows.assign(number_of_rows, 0.0);
335 const std::size_t customer_rows = std::min<std::size_t>(number_of_rows, N_SIZE - 2);
336 const std::size_t columns_to_use =
337 std::min<std::size_t>(static_cast<std::size_t>(std::max(0, numK)), pricing_columns.size());
338 for (std::size_t column_index = 0; column_index < columns_to_use; ++column_index) {
339 const Label *column = pricing_columns[column_index];
340 if (column == nullptr) continue;
341 for (uint16_t node : column->getRoute()) {
342 if (node > 0 && node < N_SIZE - 1) {
343 const std::size_t row = static_cast<std::size_t>(node - 1);
344 if (row < customer_rows) new_rows[row] += 1.0;
345 }
346 }
347 }
348
349 // Pricing routes expose customer-row coefficients. Extra cut and
350 // branching-row coefficients cannot be reconstructed from a route
351 // alone, so leave those directional components at zero.
352 subgradient.assign(number_of_rows, 0.0);
353 for (std::size_t row = 0; row < customer_rows; ++row) { subgradient[row] = model.b[row] - new_rows[row]; }
354 subgradient_norm = norm(subgradient);
355 }

376 void update_stabilization_after_pricing_optim(const ModelData &dados, const DualSolution &master_duals,
377 double lagrangian_bound,
378 const std::vector<Label *> &best_pricing_cols,
379 bool mispriced = false) {
380 align_with_master_duals(master_duals);
381 constraint_senses = dados.sense;
382
383 // The alpha and beta rules use the subgradient of a successful round;
384 // during a misprice sequence alpha follows the misprice schedule.
385 if (!mispriced && nb_misprices == 0) {
386 update_subgradient(dados, master_duals, best_pricing_cols);
387
388 // Dynamic alpha schedule (Table 1 right, with the paper's sign
389 // error corrected, as in GCG): if g_sep · (pi_out - pi_in) > 0
390 // the Lagrangian bound still increases toward pi_out, so the step
391 // was too cautious -> fdecr(alpha); otherwise -> fincr(alpha).
392 double g_dot_dir = 0.0;
393 const bool has_direction =
394 subgradient_norm > EPSILON && subgradient.size() == master_duals.size() && !cur_stab_center.empty();
395 if (has_direction) {
396 for (size_t i = 0; i < master_duals.size(); ++i) {
397 g_dot_dir += subgradient[i] * (master_duals[i] - cur_stab_center[i]);
398 }
399 }
400
401 auto fincr = [](double a) { return a + (1.0 - a) * 0.1; };
402 auto fdecr = [](double a) {
403 if (a >= 0.5) return a / 1.1;
404 return std::max(0.0, a - (1.0 - a) * 0.1);
405 };
406 if (has_direction) {
407 alpha = (g_dot_dir > 0.0) ? fdecr(alpha) : fincr(alpha);
408 alpha = std::clamp(alpha, 0.0, kAlphaMax);
409 base_alpha = std::clamp(alpha, kAlphaMin, kAlphaMax);
410 cur_alpha = alpha;
411 }
412
413 // Adaptive beta schedule (Section directional smoothing):
414 // beta = cos(gamma) between (pi_out - pi_in) and (pi_g - pi_in).
415 if (has_direction) {
416 double dir_norm_sq = 0.0;
417 double dot = 0.0;
418 for (size_t i = 0; i < master_duals.size(); ++i) {
419 const double d = master_duals[i] - cur_stab_center[i];
420 dir_norm_sq += d * d;
421 dot += d * subgradient[i];
422 }
423 const double dir_norm = std::sqrt(dir_norm_sq + EPSILON);
424 beta = std::clamp(dot / (dir_norm * subgradient_norm), 0.0, 1.0);
425 } else {
426 beta = 0.0;
427 }
428 }
429
430 // Stability center (Wentges): move to the separation point only when
431 // its Lagrangian bound beats the center's. Heuristic pricing gives no
432 // valid bound, so then the center follows each successful round.
433 const DualSolution &separation_point = smooth_dual_sol.empty() ? master_duals : smooth_dual_sol;
434 if (std::isfinite(lagrangian_bound)) {
435 if (lagrangian_bound > center_bound + kBoundTolerance) {
436 center_bound = lagrangian_bound;
437 stab_center_for_next_iteration = separation_point;
438 }
439 } else if (!mispriced) {
440 stab_center_for_next_iteration = separation_point;
441 }
442
443 if (std::isnan(alpha) || std::isinf(alpha)) {
444 stabilization_active = false;
445 cleanup();
446 }
447 }
