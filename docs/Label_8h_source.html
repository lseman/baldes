257 void materializeRoute(std::vector<uint16_t> &route) const {
258 if (parent == nullptr) {
259 const auto &stored_route = getRoute();
260 route.assign(stored_route.begin(), stored_route.end());
261 return;
262 }
263
264 const Label *root = this;
265 std::size_t depth = 0;
266 while (root->parent != nullptr) {
267 ++depth;
268 root = root->parent;
269 }
270
271 const auto &root_route = root->getRoute();
272 const std::size_t prefix_size = root_route.size();
273 route.resize(prefix_size + depth);
274 std::copy(root_route.begin(), root_route.end(), route.begin());
275
276 const Label *current = this;
277 std::size_t write = route.size();
278 while (current->parent != nullptr) {
279 route[--write] = static_cast<uint16_t>(current->node_id);
280 current = current->parent;
281 }
282 }
