18template <typename T>
19class ThreadLocalPool {
20 static thread_local std::vector<std::vector<T>> pool;
21 static thread_local size_t current_index;
22
23public:
24 std::vector<T> &acquire() {
25 if (current_index >= pool.size()) { pool.emplace_back(); }
26 return pool[current_index++];
27 }
28 void reset() { current_index = 0; }
29};

46void BucketGraph::add_arc(int from_bucket, int to_bucket, const std::vector<double> &res_inc, double cost_inc) {
47 if constexpr (D == Direction::Forward) {
48 fw_arcs.emplace_back(from_bucket, to_bucket, res_inc, cost_inc);
49 if (from_bucket >= static_cast<int>(fw_bucket_graph.size())) {
50 fw_bucket_graph.resize(static_cast<size_t>(from_bucket) + 1);
51 }
52 fw_bucket_graph[from_bucket].push_back(to_bucket);
53
54 } else if constexpr (D == Direction::Backward) {
55 bw_arcs.emplace_back(from_bucket, to_bucket, res_inc, cost_inc);
56 if (from_bucket >= static_cast<int>(bw_bucket_graph.size())) {
57 bw_bucket_graph.resize(static_cast<size_t>(from_bucket) + 1);
58 }
59 bw_bucket_graph[from_bucket].push_back(to_bucket);
60 }

63template <Direction D>
64inline int BucketGraph::get_bucket_number(int node, std::vector<double> &resource_values_vec) {
65 const size_t num_resources = options.main_resources.size();
66
67 const auto &buckets = assign_buckets<D>(fw_buckets, bw_buckets);
68
69 // Optionally adjust resource values (uncomment if you need epsilon
70 // adjustments)
71 for (size_t r = 0; r < num_resources; ++r) {
72 if constexpr (D == Direction::Forward) {
73 // resource_values_vec[r] += numericutils::eps;
74 } else { // Direction::Backward
75 // resource_values_vec[r] -= numericutils::eps;
76 }
77 }
78 auto &num_buckets_index = assign_buckets<D>(num_buckets_index_fw, num_buckets_index_bw);
79 // Start with the offset for this node’s buckets.
80 int bucket_number = num_buckets_index[node];
81 int multiplier = 1;
82
83 for (size_t r = 0; r < num_resources; ++r) {
84 // Node-specific resource bounds.
85 double node_lb = nodes[node].lb[r];
86 double node_ub = nodes[node].ub[r];
87
88 // Global bounds (if needed for computing splits).
89 double global_lb = R_min[r];
90 double global_ub = R_max[r];
91 double full_range = global_ub - global_lb;
92 double node_range = node_ub - node_lb;
93
94 // Determine the number of splits (i.e. intervals) for resource r.
95 // Keep this identical to define_buckets, including the degenerate
96 // global-range case.
97 int splits = 1;
98 if (std::fabs(full_range) > std::numeric_limits<double>::epsilon()) {
99 splits =
100 std::max(1, static_cast<int>(std::round((node_range / full_range) * intervals[r].interval)));
101 }
102 double interval_width =
103 (splits == 0 || std::fabs(node_range) < std::numeric_limits<double>::epsilon()) ? 1.0 : node_range / splits;
104
105 int pos = 0;
106 if constexpr (D == Direction::Forward) {
107 if (std::fabs(interval_width) < std::numeric_limits<double>::epsilon()) {
108 pos = 0;
109 } else {
110 pos = std::clamp(static_cast<int>(std::floor((resource_values_vec[r] - node_lb) / interval_width)), 0,
111 splits - 1);
112 }
113 } else {
114 if (std::fabs(interval_width) < std::numeric_limits<double>::epsilon()) {
115 pos = 0;
116 } else {
117 pos = std::clamp(static_cast<int>(std::floor((node_ub - resource_values_vec[r]) / interval_width)), 0,
118 splits - 1);
119 }
120 }
121
122 bucket_number += pos * multiplier;
123 multiplier *= splits;
124 }
125
126 // Buckets contain only the bucketed (main) dimensions, whereas labels may
127 // carry additional resources. Validate only the dimensions represented by
128 // the bucket to avoid indexing past its compact bounds.
129 bool bucket_contains_values = bucket_number >= 0 && bucket_number < static_cast<int>(buckets.size());
130 if (bucket_contains_values) {
131 const auto &bucket = buckets[bucket_number];
132 bucket_contains_values =
133 bucket.lb.size() == num_resources && bucket.ub.size() == num_resources &&
134 resource_values_vec.size() >= num_resources;
135 for (size_t r = 0; r < num_resources && bucket_contains_values; ++r) {
136 bucket_contains_values = !numericutils::lt(resource_values_vec[r], bucket.lb[r]) &&
137 !numericutils::gt(resource_values_vec[r], bucket.ub[r]);
138 }
139 }
140 if (unlikely(!bucket_contains_values)) {
141 // print resource_values_vec
142 for (auto val : resource_values_vec) { fmt::print("{}\n", val); }
143 // Print bucket bounds only when the computed index itself is valid.
144 if (bucket_number >= 0 && bucket_number < static_cast<int>(buckets.size()) &&
145 !buckets[bucket_number].lb.empty() && !buckets[bucket_number].ub.empty()) {
146 fmt::print("{}\n", buckets[bucket_number].lb[0]);
147 fmt::print("{}\n", buckets[bucket_number].ub[0]);
148 }
149
150 std::throw_with_nested(std::runtime_error("Resource values not contained in bucket"));
151 }
152
153 return bucket_number;
154}

165template <Direction D>
166// A constant representing the number of splits for a node that spans the full
167// range. For example, if a node covers the entire full range, it will get
168// FULL_RANGE_SPLITS buckets.
169void BucketGraph::define_buckets() {
170 // 1. Save the old buckets and fixed status.
171 std::vector<Bucket> old_buckets;
172 std::vector<bool> was_fixed;
173 auto &buckets = assign_buckets<D>(fw_buckets, bw_buckets);
174 size_t old_size = buckets.size();
175
176 // Map to store the fixed range (min_lb and max_ub) for each node pair.
177 ankerl::unordered_dense::map<int,
178 ankerl::unordered_dense::map<int, std::pair<std::vector<double>, std::vector<double>>>>
179 node_pair_fixed_ranges;
180
181 if (old_size > 0) {
182 old_buckets = buckets;
183 was_fixed.resize(old_size * old_size, false);
184 auto &old_bitmap = assign_buckets<D>(fw_fixed_buckets_bitmap, bw_fixed_buckets_bitmap);
185
186 // Mark fixed bucket pairs.
187 for (size_t i = 0; i < old_size; ++i) {
188 for (size_t j = 0; j < old_size; ++j) {
189 size_t pos = i * old_size + j;
190 if (pos / 64 < old_bitmap.size()) {
191 was_fixed[pos] = ((old_bitmap[pos / 64] & (1ULL << (pos % 64))) != 0);
192 }
193 }
194 }
195
196 const int num_intervals = options.main_resources.size();
197 // For each fixed bucket pair, update the fixed range for the
198 // corresponding node pair.
199 for (size_t i = 0; i < old_size; ++i) {
200 for (size_t j = 0; j < old_size; ++j) {
201 if (!was_fixed[i * old_size + j]) continue;
202 const auto &bucket_from = old_buckets[i];
203 const auto &bucket_to = old_buckets[j];
204 int from_node = bucket_from.node_id;
205 int to_node = bucket_to.node_id;
206 // Initialize if necessary.
207 if (node_pair_fixed_ranges[from_node].find(to_node) == node_pair_fixed_ranges[from_node].end()) {
208 node_pair_fixed_ranges[from_node][to_node] = {
209 std::vector<double>(num_intervals, std::numeric_limits<double>::max()),
210 std::vector<double>(num_intervals, std::numeric_limits<double>::lowest())};
211 }
212 auto &[min_lbs, max_ubs] = node_pair_fixed_ranges[from_node][to_node];
213 for (int r = 0; r < num_intervals; ++r) {
214 min_lbs[r] = std::min(min_lbs[r], bucket_from.lb[r]);
215 max_ubs[r] = std::max(max_ubs[r], bucket_from.ub[r]);
216 }
217 }
218 }
219 } // End if old buckets exist.
220
221 // 2. Clear the current buckets and fixed bitmap.
222 buckets.clear();
223 auto &fixed_buckets_bitmap = assign_buckets<D>(fw_fixed_buckets_bitmap, bw_fixed_buckets_bitmap);
224 fixed_buckets_bitmap.clear();
225
226 // 3. Prepare variables and containers for new bucket definition.
227 const int num_intervals = options.main_resources.size();
228 std::vector<double> total_ranges(num_intervals); // unused in current code
229 std::vector<double> base_intervals(num_intervals); // unused in current code
230
231 auto &num_buckets = assign_buckets<D>(num_buckets_fw, num_buckets_bw);
232 auto &num_buckets_index = assign_buckets<D>(num_buckets_index_fw, num_buckets_index_bw);
233 auto &buckets_size = assign_buckets<D>(fw_buckets_size, bw_buckets_size);
234 auto &bucket_splits = assign_buckets<D>(fw_bucket_splits, bw_bucket_splits);
235
236 const size_t num_nodes = nodes.size();
237 num_buckets.resize(num_nodes);
238 num_buckets_index.resize(num_nodes);
239
240 // Lambda to calculate an interval for one dimension. Backward buckets are
241 // deliberately emitted in the paper's kappa order:
242 // kappa = 0 -> interval ending at u_v,
243 // kappa = 1 -> next lower interval, ...
244 // This keeps component-wise smaller buckets equal to smaller local bucket
245 // positions in both forward and backward arrays.
246 auto calculate_interval = [&](double lb, double ub, double base_interval, int pos, int max_intervals,
247 bool is_forward) -> std::pair<double, double> {
248 double start, end;
249 if (is_forward) {
250 start = (pos == 0) ? lb : lb + pos * base_interval;
251 end = (pos == max_intervals - 1) ? ub : lb + (pos + 1) * base_interval;
252 } else {
253 start = (pos == max_intervals - 1) ? lb : ub - (pos + 1) * base_interval;
254 end = (pos == 0) ? ub : ub - pos * base_interval;
255 }
256 return {start, end};
257 };
258
259 int cum_sum = 0;
260 int bucket_index = 0;
261 std::vector<double> interval_start(num_intervals);
262 std::vector<double> interval_end(num_intervals);
263 std::vector<int> pos(num_intervals,
264 0); // combination indices for multi-dimensions
265
266 // 4. Process each node to define new buckets.
267 for (const auto &vrp_node : nodes) {
268 std::vector<int> node_split_counts(num_intervals);
269 std::vector<double> node_base_interval(num_intervals);
270 // Determine the number of splits and base interval for each
271 // resource dimension.
272 for (int r = 0; r < num_intervals; ++r) {
273 double full_range = R_max[r] - R_min[r];
274 double node_range = vrp_node.ub[r] - vrp_node.lb[r];
275 int splits = 1;
276 if (std::fabs(full_range) > std::numeric_limits<double>::epsilon()) {
277 double splits_d = (node_range * intervals[r].interval) / full_range;
278 splits = std::max(1, static_cast<int>(std::round(splits_d)));
279 }
280 node_split_counts[r] = splits;
281 node_base_interval[r] = (vrp_node.ub[r] - vrp_node.lb[r]) / static_cast<double>(splits);
282 }
283 // Optionally store node-specific split counts.
284 bucket_splits[vrp_node.id] = node_split_counts;
285
286 int n_buckets = 0;
287 if (num_intervals == 1) {
288 // Single-dimensional case.
289 for (int j = 0; j < node_split_counts[0]; ++j) {
290 auto [int_start, int_end] = calculate_interval(vrp_node.lb[0], vrp_node.ub[0], node_base_interval[0], j,
291 node_split_counts[0], D == Direction::Forward);
292 if constexpr (D == Direction::Backward)
293 int_start = std::max(int_start, vrp_node.lb[0]);
294 else
295 int_end = std::min(int_end, vrp_node.ub[0]);
296 buckets.push_back(Bucket(vrp_node.id, std::vector<double>{int_start}, std::vector<double>{int_end}));
297 n_buckets++;
298 cum_sum++;
299 }
300 } else {
301 // Multi-dimensional: iterate over all combinations.
302 fmt::print("Multiple intervals\n");
303 std::fill(pos.begin(), pos.end(), 0);
304 while (true) {
305 for (int r = 0; r < num_intervals; ++r) {
306 auto [int_start, int_end] =
307 calculate_interval(vrp_node.lb[r], vrp_node.ub[r], node_base_interval[r], pos[r],
308 node_split_counts[r], D == Direction::Forward);
309 interval_start[r] = int_start;
310 interval_end[r] = int_end;
311 if constexpr (D == Direction::Backward)
312 interval_start[r] = std::max(interval_start[r], R_min[r]);
313 else
314 interval_end[r] = std::min(interval_end[r], R_max[r]);
315 }
316 buckets.push_back(Bucket(vrp_node.id, interval_start, interval_end));
317 n_buckets++;
318 cum_sum++;
319
320 // Generate next combination.
321 int i = 0;
322 while (i < num_intervals) {
323 ++pos[i];
324 if (pos[i] < node_split_counts[i])
325 break;
326 else {
327 pos[i] = 0;
328 ++i;
329 }
330 }
331 if (i == num_intervals) break;
332 }
333 }
334 num_buckets[vrp_node.id] = n_buckets;
335 num_buckets_index[vrp_node.id] = cum_sum - n_buckets;
336 if constexpr (D == Direction::Forward) { n_segments = (n_buckets + 63) >> 6; }
337 }
338
339 // 5. Precompute RC2 bucket metadata for whole-bucket prune.
340 // Forward dominance needs candidate rc2 <= label rc2, so keep a prefix
341 // minimum of bucket lower bounds. Backward buckets are reversed, so
342 // candidate rc2 >= label rc2 also lives in the prefix.
343 auto &rc2_bin = assign_buckets<D>(fw_rc2_bin, bw_rc2_bin);
344 auto &rc2_till_this_bin = assign_buckets<D>(fw_rc2_till_this_bin, bw_rc2_till_this_bin);
345 rc2_bin.assign(num_nodes, {});
346 rc2_till_this_bin.assign(num_nodes, {});
347 for (int node_id = 0; node_id < static_cast<int>(num_nodes); ++node_id) {
348 int count = num_buckets[node_id];
349 rc2_bin[node_id].assign(count, 0.0);
350 rc2_till_this_bin[node_id].assign(count, 0.0);
351 if (count == 0) continue;
352
353 for (int local_b = 0; local_b < count; ++local_b) {
354 int global_b = num_buckets_index[node_id] + local_b;
355 const auto &bucket = buckets[global_b];
356 double rc2_value = 0.0;
357 if constexpr (D == Direction::Forward) {
358 if (bucket.lb.size() > 1) { rc2_value = bucket.lb[1]; }
359 } else {
360 if (bucket.ub.size() > 1) { rc2_value = bucket.ub[1]; }
361 }
362 rc2_bin[node_id][local_b] = rc2_value;
363 if (local_b == 0) {
364 rc2_till_this_bin[node_id][local_b] = rc2_value;
365 } else {
366 if constexpr (D == Direction::Forward) {
367 rc2_till_this_bin[node_id][local_b] = std::min(rc2_till_this_bin[node_id][local_b - 1], rc2_value);
368 } else {
369 rc2_till_this_bin[node_id][local_b] = std::max(rc2_till_this_bin[node_id][local_b - 1], rc2_value);
370 }
371 }
372 }
373 }
374
375 // 6. Process fixed bucket ranges (if available).
376 if (!node_pair_fixed_ranges.empty()) {
377 size_t new_size = buckets.size();
378 size_t required_bitmap_words = std::max(size_t(1), ((new_size * new_size) + 63) / 64);
379 fixed_buckets_bitmap.clear();
380 fixed_buckets_bitmap.resize(required_bitmap_words, 0);
381 for (size_t from = 0; from < new_size; ++from) {
382 const auto &bucket_from = buckets[from];
383 int from_node = bucket_from.node_id;
384 if (node_pair_fixed_ranges.find(from_node) == node_pair_fixed_ranges.end()) continue;
385 for (size_t to = 0; to < new_size; ++to) {
386 const auto &bucket_to = buckets[to];
387 int to_node = bucket_to.node_id;
388 if (node_pair_fixed_ranges[from_node].find(to_node) == node_pair_fixed_ranges[from_node].end())
389 continue;
390 const auto &range = node_pair_fixed_ranges[from_node][to_node];
391 bool in_range = true;
392 for (int r = 0; r < options.main_resources.size(); ++r) {
393 if (bucket_from.lb[r] < range.first[r] || bucket_from.ub[r] > range.second[r]) {
394 in_range = false;
395 break;
396 }
397 }
398 if (in_range) {
399 size_t bit_pos = from * new_size + to;
400 size_t word_idx = bit_pos / 64;
401 if (word_idx < fixed_buckets_bitmap.size()) {
402 fixed_buckets_bitmap[word_idx] |= (1ULL << (bit_pos % 64));
403 }
404 }
405 }
406 }
407 }
408
409 // 6. Finalize: record the total bucket count.
410 buckets_size = buckets.size();
411}

422template <Direction D>
423int BucketGraph::get_opposite_bucket_number(int current_bucket_index, std::vector<double> &inc) {
424 // Retrieve the current bucket based on the current direction.
425 const auto &current_bucket =
426 (D == Direction::Forward) ? fw_buckets[current_bucket_index] : bw_buckets[current_bucket_index];
427 const int node = current_bucket.node_id;
428 const auto &theNode = nodes[node];
429
430 // Pre-compute the reference point for the given node using the main
431 // resources.
432 const size_t num_resources = options.main_resources.size();
433 // Optionally, add an assertion or check that inc has the expected size.
434 assert(inc.size() == num_resources && theNode.lb.size() == num_resources && theNode.ub.size() == num_resources);
435
436 std::vector<double> reference_point(num_resources);
437
438 if constexpr (D == Direction::Forward) {
439 // For Forward direction:
440 // Use the maximum of inc and the node's lower bounds.
441 // This ensures we do not fall below the allowed lower bound.
442 for (size_t i = 0; i < num_resources; ++i) {
443 reference_point[i] = std::max(inc[i], theNode.lb[i]);
444 // Optionally, adjust with epsilon if needed:
445 // reference_point[i] += numericutils::eps;
446 }
447 // Get the opposite bucket using the Backward tree.
448 return get_bucket_number<Direction::Backward>(node, reference_point);
449 } else {
450 // For Backward direction:
451 // Use the minimum of inc and the node's upper bounds.
452 // This ensures we do not exceed the allowed upper bound.
453 for (size_t i = 0; i < num_resources; ++i) {
454 reference_point[i] = std::min(inc[i], theNode.ub[i]);
455 // Optionally, adjust with epsilon if needed:
456 // reference_point[i] -= numericutils::eps;
457 }
458 // Get the opposite bucket using the Forward tree.
459 return get_bucket_number<Direction::Forward>(node, reference_point);
460 }
461}

473template <Stage S>
474void BucketGraph::bucket_fixing() {
475 // Only execute if not already fixed.
476 if (!fixed) {
477 fmt::print("\033[34m_STARTING BUCKET FIXING PROCEDURE\033[0m\n");
478 fixed = true;
479
480 // Compute gap and check feasibility (if gap is negative, exit
481 // early).
482 gap = std::ceil(incumbent - (relaxation + std::min(0.0, min_red_cost)));
483 if (gap < 0) {
484 fmt::print("\033[34m_BUCKET FIXING PROCEDURE CAN'T BE EXECUTED DUE TO "
485 "GAP\033[0m\n");
486 fmt::print("gap: {}\n", gap);
487 return;
488 }
489
490 // Initialize common structures, arc scores, warm labels, etc.
491 common_initialization();
492
493 // Initialize forward and backward c_bar vectors.
494 std::vector<double> forward_cbar(fw_buckets.size(), std::numeric_limits<double>::infinity());
495 std::vector<double> backward_cbar(bw_buckets.size(), std::numeric_limits<double>::infinity());
496
497 // Run labeling algorithms for Stage Four with full mode.
498 run_labeling_algorithms<Stage::Four, Full::Full>(forward_cbar, backward_cbar);
499 // Save computed c_bar values.
500 fw_c_bar = forward_cbar;
501 bw_c_bar = backward_cbar;
502
503 // Print info about bucket arc elimination.
504 print_info("performing bucket arc elimination with theta = {}\n", gap);
505
506 // Run forward and backward sections in parallel.
507 PARALLEL_SECTIONS(
508 work, bi_sched,
509 SECTION {
510 // Section 1: Process forward direction.
511 BucketArcElimination<Direction::Forward>(gap);
512 ObtainJumpBucketArcs<Direction::Forward>();
513 },
514 SECTION {
515 // Section 2: Process backward direction.
516 BucketArcElimination<Direction::Backward>(gap);
517 ObtainJumpBucketArcs<Direction::Backward>();
518 });
519
520 // Generate arcs after elimination.
521 // generate_arcs();
522 fmt::print("\033[34m_BUCKET FIXING PROCEDURE FINISHED\033[0m\n");
523
524 just_fixed = true;
525 return;
526 }
527 just_fixed = false;
528}

537template <Stage S>
538void BucketGraph::heuristic_fixing() {
539 if (fixed) { return; }
540
541 // Reset and initialize state.
542 reset_pool();
543 reset_fixed();
544 common_initialization();
545
546 // Pre-allocate c-bar vectors with infinity.
547 std::vector<double> forward_cbar(fw_buckets.size(), std::numeric_limits<double>::infinity());
548 std::vector<double> backward_cbar(bw_buckets.size(), std::numeric_limits<double>::infinity());
549
550 gap = std::ceil(incumbent - (relaxation + std::min(0.0, min_red_cost)));
551
552 // Run the appropriate labeling algorithms based on stage
553 if (S == Stage::Three) {
554 // Run the labeling algorithms for Stage::Two with Full::Partial mode.
555 run_labeling_algorithms<Stage::Two, Full::Partial>(forward_cbar, backward_cbar);
556 } else if (S == Stage::Four) {
557 fmt::print("\033[34m_STARTING ARC FIXING PROCEDURE\033[0m\n");
558 fmt::print("gap: {}\n", gap);
559 // Run the labeling algorithms for Stage::Four with Full::Full mode.
560 run_labeling_algorithms<Stage::Four, Full::Full>(forward_cbar, backward_cbar);
561 }
562
563 // Prepare containers for grouping labels by node.
564 const size_t num_nodes = nodes.size();
565
566 // Preallocate vectors with nullptr to avoid resizing
567 std::vector<const Label *> min_fw_labels(num_nodes, nullptr);
568 std::vector<const Label *> min_bw_labels(num_nodes, nullptr);
569
570 // Find minimum cost label for each node directly from buckets
571 // without creating intermediate label maps
572 for (auto &bucket : fw_buckets) {
573 for (const Label *label : bucket.get_labels()) {
574 if (label == nullptr || label->is_dominated) continue;
575 const size_t node_id = label->node_id;
576 if (!min_fw_labels[node_id] || label->cost < min_fw_labels[node_id]->cost) {
577 min_fw_labels[node_id] = label;
578 }
579 }
580 }
581
582 for (auto &bucket : bw_buckets) {
583 for (const Label *label : bucket.get_labels()) {
584 if (label == nullptr || label->is_dominated) continue;
585 const size_t node_id = label->node_id;
586 if (!min_bw_labels[node_id] || label->cost < min_bw_labels[node_id]->cost) {
587 min_bw_labels[node_id] = label;
588 }
589 }
590 }
591
592 // Pre-compute the index of the "time" resource.
593 const size_t time_index =
594 std::distance(options.resources.begin(), std::find(options.resources.begin(), options.resources.end(), "time"));
595
596 // Precompute lookup tables for common values
597 std::vector<double> fw_time_resources(num_nodes, 0.0);
598 std::vector<double> bw_time_resources(num_nodes, 0.0);
599 std::vector<double> fw_costs(num_nodes, 0.0);
600 std::vector<double> bw_costs(num_nodes, 0.0);
601
602 for (size_t i = 0; i < num_nodes; ++i) {
603 if (min_fw_labels[i]) {
604 fw_time_resources[i] = min_fw_labels[i]->resources[time_index];
605 fw_costs[i] = min_fw_labels[i]->cost;
606 }
607 if (min_bw_labels[i]) {
608 bw_time_resources[i] = min_bw_labels[i]->resources[time_index];
609 bw_costs[i] = min_bw_labels[i]->cost;
610 }
611 }
612
613 // Cache node data to avoid repeated lookups
614 std::vector<double> node_durations(num_nodes);
615 std::vector<std::vector<double>> node_consumptions(num_nodes, std::vector<double>(options.resources.size()));
616
617 for (size_t i = 0; i < num_nodes; ++i) {
618 const VRPNode &node = nodes[i];
619 node_durations[i] = node.duration;
620 for (size_t r = 0; r < options.resources.size(); ++r) { node_consumptions[i][r] = node.consumption[r]; }
621 }
622
623 size_t num_fixes = 0;
624 const double gap_threshold = gap; // Cache this value
625
626 // Build a list of candidate node pairs to check
627 struct NodePair {
628 size_t i;
629 size_t j;
630 };
631
632 std::vector<NodePair> candidate_pairs;
633 candidate_pairs.reserve(num_nodes * num_nodes / 4); // Estimate: 25% of pairs will be candidates
634
635 for (size_t i = 0; i < num_nodes; ++i) {
636 if (!min_fw_labels[i]) continue;
637
638 for (size_t j = 0; j < num_nodes; ++j) {
639 if (i == j || !min_bw_labels[j]) continue;
640 candidate_pairs.push_back({i, j});
641 }
642 }
643
644 // Process candidate pairs in batches to improve cache locality
645 const size_t BATCH_SIZE = 256;
646 std::vector<std::pair<size_t, size_t>> arcs_to_fix;
647 arcs_to_fix.reserve(BATCH_SIZE);
648
649 for (size_t start = 0; start < candidate_pairs.size(); start += BATCH_SIZE) {
650 const size_t end = std::min(start + BATCH_SIZE, candidate_pairs.size());
651 arcs_to_fix.clear();
652
653 for (size_t idx = start; idx < end; ++idx) {
654 const auto &pair = candidate_pairs[idx];
655 const size_t i = pair.i;
656 const size_t j = pair.j;
657
658 const double cost = getcij(min_fw_labels[i]->node_id, min_bw_labels[j]->node_id);
659
660 // Early rejection using precomputed time resources (often most
661 // restrictive)
662 if (options.resources[time_index] == "time") {
663 if (fw_time_resources[i] + cost + node_durations[min_fw_labels[i]->node_id] > bw_time_resources[j]) {
664 continue;
665 }
666 }
667
668 // Check all other resources
669 bool violated = false;
670 const size_t fw_node_id = min_fw_labels[i]->node_id;
671
672 for (size_t r = 0; r < options.resources.size(); ++r) {
673 if (r != time_index) {
674 if (min_fw_labels[i]->resources[r] + node_consumptions[fw_node_id][r] >
675 min_bw_labels[j]->resources[r]) {
676 violated = true;
677 break;
678 }
679 }
680 }
681
682 // If all resource constraints are satisfied and the cost exceeds
683 // the gap, fix the arc
684 if (!violated && (fw_costs[i] + cost + bw_costs[j] > gap_threshold)) {
685 arcs_to_fix.emplace_back(i, j);
686 ++num_fixes;
687 }
688 }
689
690 // Fix arcs in batch
691 for (const auto &[i, j] : arcs_to_fix) { fix_arc(i, j); }
692 }
693
694 if (S == Stage::Four) {
695 fmt::print("Applied {} arc fixes\n", num_fixes);
696 fmt::print("\033[34m_ARC FIXING PROCEDURE FINISHED\033[0m\n");
697 }

700template <Symmetry SYM>
701void BucketGraph::set_adjacency_list() {
702 // === Step 1: Clear Existing Arcs from Each Node ===
703 for (auto &node : nodes) { node.clear_arcs(); }
704
705 // === Step 2: MST-Based Clustering ===
706 MST mst_solver(nodes, [this](int from, int to) { return getcij(from, to); });
707 auto mst = mst_solver.compute_mst();
708
709 // Collect and sort MST edge weights.
710 std::vector<double> edge_weights;
711 edge_weights.reserve(mst.size());
712 for (const auto &[weight, from, to] : mst) { edge_weights.push_back(weight); }
713 pdqsort(edge_weights.begin(), edge_weights.end());
714
715 // Compute theta as the 90th percentile (scaled down by 100).
716 size_t p90_index = static_cast<size_t>(0.9 * edge_weights.size());
717 double theta = (edge_weights[p90_index]) / 100.0;
718 print_info("Computed theta: {}\n", theta);
719
720 // Cluster nodes and build mapping: job_id -> cluster_id.
721 auto clusters = mst_solver.cluster(theta);
722 print_info("Number of clusters: {}\n", clusters.size());
723 std::vector<int> job_to_cluster(nodes.size(), -1);
724 for (int cluster_id = 0; cluster_id < clusters.size(); ++cluster_id) {
725 for (int job : clusters[cluster_id]) { job_to_cluster[job] = cluster_id; }
726 }
727
728 // === Step 3: Precompute Resource Information ===
729 // Find the index for the "time" resource.
730 auto time_it = std::find(options.resources.begin(), options.resources.end(), "time");
731 const int time_resource_idx =
732 (time_it != options.resources.end()) ? std::distance(options.resources.begin(), time_it) : -1;
733
734 // === Step 4: Process Arcs for Each Node ===
735
736 // Helper lambda to compute resource increments.
737 // Depending on symmetry, consumption is either taken as-is or averaged.
738 auto compute_resource_increments = [&](const VRPNode &from_node, const VRPNode &to_node, double travel_cost,
739 std::vector<double> &inc) {
740 const int nres = options.resources.size();
741 if constexpr (SYM == Symmetry::Asymmetric) {
742 for (int r = 0; r < nres; ++r) {
743 inc[r] = (r == time_resource_idx) ? travel_cost + from_node.duration : from_node.consumption[r];
744 }
745 } else {
746 for (int r = 0; r < nres; ++r) {
747 inc[r] = (r == time_resource_idx) ? (from_node.duration / 2.0 + travel_cost + to_node.duration / 2.0)
748 : ((from_node.consumption[r] + to_node.consumption[r]) / 2.0);
749 }
750 }
751 };
752
753 // Lambda to add forward and reverse arcs from a given node.
754 auto add_arcs_for_node = [&](const VRPNode &curr_node) {
755 // Containers to accumulate arcs before insertion.
756 using ArcTuple = std::tuple<double, int, std::vector<double>, double>;
757 std::vector<ArcTuple> forward_arcs;
758 std::vector<ArcTuple> reverse_arcs;
759 forward_arcs.reserve(nodes.size());
760 reverse_arcs.reserve(nodes.size());
761
762 // Iterate over candidate next nodes.
763 for (const auto &next_node : nodes) {
764 // Skip if the candidate is a depot, the same node, or a pstep
765 // depot.
766 if (next_node.id == options.depot || curr_node.id == next_node.id ||
767 (options.pstep && (next_node.id == options.pstep_depot || next_node.id == options.pstep_end_depot))) {
768 continue;
769 }
770
771 double travel_cost = getcij(curr_node.id, next_node.id);
772 double cost_inc = travel_cost - next_node.cost;
773
774 // Skip self-loops (or if bucket indexes match; here using node
775 // id as bucket id).
776 if (curr_node.id == next_node.id) continue;
777
778 // Compute resource increments.
779 std::vector<double> res_inc(options.resources.size());
780 compute_resource_increments(curr_node, next_node, travel_cost, res_inc);
781
782 // Check feasibility: ensure resource consumption does not
783 // exceed next node's upper bound.
784 bool feasible = true;
785 for (int r = 0; r < options.resources.size(); ++r) {
786 if (numericutils::gt(curr_node.lb[r] + res_inc[r], next_node.ub[r])) {
787 feasible = false;
788 break;
789 }
790 }
791 if (!feasible) continue;
792
793 // Compute priorities based on cluster membership.
794 bool same_cluster = (job_to_cluster[curr_node.id] == job_to_cluster[next_node.id]);
795 double forward_priority = (same_cluster ? 5.0 : 1.0) + 1.E-5 * next_node.start_time;
796 double reverse_priority = (same_cluster ? 1.0 : 5.0) + 1.E-5 * curr_node.start_time;
797
798 // Collect arcs for forward and reverse directions.
799 forward_arcs.emplace_back(forward_priority, next_node.id, res_inc, cost_inc);
800 reverse_arcs.emplace_back(reverse_priority, next_node.id, res_inc, cost_inc);
801 }
802
803 // Insert the arcs into the corresponding nodes.
804 for (const auto &[priority, to_node, res_inc, cost_inc] : forward_arcs) {
805 nodes[curr_node.id].template add_arc<Direction::Forward>(curr_node.id, to_node, res_inc, cost_inc,
806 priority);
807 }
808 for (const auto &[priority, from_node, res_inc, cost_inc] : reverse_arcs) {
809 nodes[from_node].template add_arc<Direction::Backward>(from_node, curr_node.id, res_inc, cost_inc,
810 priority);
811 }
812 };
813
814 // Process each node, skipping the end depot.
815 for (const auto &node : nodes) {
816 if (node.id != options.end_depot) { add_arcs_for_node(node); }
817 }
818}

826void BucketGraph::common_initialization() {
827 // Retrieve active cuts and build a vector of ActiveCutInfo.
828 auto &cutter = cut_storage;
829 const auto active_cuts = cutter->getActiveCuts();
830 std::vector<ActiveCutInfo> active_cuts_info;
831 active_cuts_info.reserve(active_cuts.size());
832 for (const auto &cut : active_cuts) { active_cuts_info.push_back(cut); }
833
834 // --- Sort arcs by scores for each node ---
835 auto &arc_scores = assign_buckets<D>(fw_arc_scores, bw_arc_scores);
836 for (auto &node : nodes) {
837 if (!arc_scores[node.id].empty()) { node.sort_arcs_by_scores<D>(arc_scores[node.id], nodes, active_cuts_info); }
838 arc_scores[node.id].clear();
839 }
840
841 auto &warm_labels = assign_buckets<D>(fw_warm_labels, bw_warm_labels);
842 auto &buckets_size = assign_buckets<D>(fw_buckets_size, bw_buckets_size);
843 auto &buckets = assign_buckets<D>(fw_buckets, bw_buckets);
844 auto &c_bar = assign_buckets<D>(fw_c_bar, bw_c_bar);
845 auto &bucket_splits = assign_buckets<D>(fw_bucket_splits, bw_bucket_splits);
846
847 // --- Initialize c_bar vector ---
848 c_bar.resize(buckets_size, std::numeric_limits<double>::infinity());
849
850 // --- Initialize dominance check counters ---
851 if constexpr (Direction::Forward == D) {
852 dominance_checks_per_bucket.assign(buckets_size + 1, 0);
853 non_dominated_labels_per_bucket = 0;
854 regen_dominance_checks_fw = 0;
855 } else {
856 non_dominated_labels_per_bucket_bw = 0;
857 regen_dominance_checks_bw = 0;
858 }
859
860 // --- Clear all buckets ---
861 for (size_t b = 0; b < buckets_size; b++) { buckets[b].clear(); }
862
863 // --- Warm start processing: sort warm_labels and compute reduced costs
864 // ---
865 if (options.warm_start && !just_fixed && !warm_labels.empty()) {
866 pdqsort(warm_labels.begin(), warm_labels.end(),
867 [](const WarmLabelState &a, const WarmLabelState &b) { return a.cost < b.cost; });
868 const size_t process_size =
869 std::min(warm_labels.size(), static_cast<size_t>(options.n_warm_start * warm_labels.size()));
870 for (size_t i = 0; i < process_size; ++i) {
871 auto new_label = compute_red_cost(warm_labels[i], D == Direction::Forward);
872 if (new_label != nullptr) {
873 if constexpr (D == Direction::Forward) {
874 fw_buckets[new_label->vertex].add_sorted_label(new_label);
875 } else {
876 bw_buckets[new_label->vertex].add_sorted_label(new_label);
877 }
878 }
879 }
880 warm_labels.clear();
881 }
882
883 const int depot_id = (D == Direction::Forward) ? options.depot : options.end_depot;
884 auto &depot_node = nodes[depot_id];
885
886 // Labels carry every resource, even when only a subset is bucketed.
887 // In particular, backward auxiliary resources must start at their upper
888 // bounds rather than retaining the pool's zero-initialized state.
889 std::vector<double> initial_resources(options.resources.size(), 0.0);
890 for (size_t r = 0; r < initial_resources.size(); ++r) {
891 initial_resources[r] = (D == Direction::Forward) ? depot_node.lb[r] : depot_node.ub[r];
892 }
893
894 const int initial_bucket = get_bucket_number<D>(depot_id, initial_resources);
895 auto &label_pool = assign_buckets<D>(label_pool_fw, label_pool_bw);
896 auto depot_label = label_pool->acquire(initial_bucket);
897 depot_label->initialize(initial_bucket, 0.0, initial_resources, depot_id);
898 depot_label->is_extended = false;
899 depot_label->addNode(depot_id);
900 set_node_visited(depot_label->visited_bitmap, depot_id);
901 SRC_MODE_BLOCK(depot_label->SRCmap.assign(cut_storage->activeSize(), 0);)
902 buckets[initial_bucket].add_label(depot_label);
903 buckets[initial_bucket].node_id = depot_id;
904}
